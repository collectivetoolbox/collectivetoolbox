// Helper that accumulates unmappables and knows how to flush them to output.
struct UnmappableBuffer {
    buf: Vec<u32>,
    found_any: bool,
    basenb_enabled: bool,
    basenb_fragment_enabled: bool,
}

impl UnmappableBuffer {
    fn new(basenb_enabled: bool, basenb_fragment_enabled: bool) -> Self {
        Self { buf: Vec::new(), found_any: false, basenb_enabled, basenb_fragment_enabled }
    }

    fn push(&mut self, dc: u32) { self.buf.push(dc); }

    fn is_empty(&self) -> bool { self.buf.is_empty() }

    // flush when forced or when there are accumulated items.
    fn flush_if_needed(
        &mut self,
        out: &mut Vec<u8>,
        force: bool,
    ) -> Result<()> {
        if !self.basenb_enabled || (self.buf.is_empty() && !force) { return Ok(()); }

        if !self.found_any && !self.basenb_fragment_enabled {
            out.extend(get_armored_utf8_embedded_start_uuid());
        }
        self.found_any = true;

        for &dc in &self.buf {
            let packed = pack32(dc)?;
            let encoded = byte_array_to_base17b_utf8(&packed)?;
            out.extend(encoded);
        }
        self.buf.clear();
        Ok(())
    }

    fn close_if_needed(&self, out: &mut Vec<u8>) {
        if self.basenb_enabled && self.found_any && !self.basenb_fragment_enabled {
            out.extend(get_armored_utf8_embedded_end_uuid());
        }
    }
}

}

    if dc_array[start_idx] != DC_START_ENCAPSULATION_UTF8 { return Ok(EncapsulationResult::NotStarted); }

    let len = dc_array.len();
    let mut j = start_idx + 1;
    let mut truncated = true;
    while j < len {
        let cur = dc_array[j];
        if cur == DC_END_ENCAPSULATION_UTF8 {
            truncated = false;
            break;
        }
        if !is_dc_base64_encapsulation_character(cur) {
            truncated = true;
            break;
        }
        j += 1;
    }

    if truncated {
        // determine subsequence to retry (exclude invalid char if not end-of-array)
        let end_exclusive = j.min(len);
        if j >= len {
            log.warn(&format!(
                "Truncated encapsulated UTF-8 sequence at index {} (missing end marker)",
                start_idx
            ));
        } else {
            log.warn(&format!(
                "Invalid character {} inside encapsulated UTF-8 sequence starting at index {} (treating as truncated)",
                dc_array[j], start_idx
            ));
        }

            subseq: dc_array[start_idx..end_exclusive].to_vec(),
            consumed: end_exclusive - start_idx,
        })

        // valid: decode inner

        match dc_encapsulated_raw_to_bytes(inner) {
            Ok(bytes) => {
                if debug {

                        "Decoded encapsulated UTF-8 sequence at {}..{} ({} inner dcs, {} bytes)",
                        start_idx, j, inner.len(), bytes.len()

                }
                Ok(EncapsulationResult::Handled { bytes, consumed: (j - start_idx) + 1 })
            }
            Err(e) => {

                    subseq: dc_array[start_idx..=j].to_vec(),
                    consumed: (j - start_idx) + 1,

pub fn dca_to_utf8(dc_array: &[u32], settings: &UTF8FormatSettings) -> Result<(Vec<u8>, FormatLog)> {
    let mut log = FormatLog::default();
    let debug = settings.debug;

    let mut out = Vec::new();
    let mut unm = UnmappableBuffer::new(settings.dc_basenb_enabled, settings.dc_basenb_fragment_enabled);

    let mut escape_next = false;
    let mut escape_this = false;

    let len = dc_array.len();
    let mut i = 0usize;
    while i < len {
        let dc = dc_array[i];

        // update escape flags
        if escape_next { escape_next = false; escape_this = true; }
        if dc == DC_ESCAPE_NEXT { escape_next = true; }

        // try encapsulation
        match try_handle_encapsulation(dc_array, i, settings, debug, &mut log)? {
            EncapsulationResult::NotStarted => { /* continue to normal mapping below */ }
            EncapsulationResult::Handled { bytes, consumed } => {
                unm.flush_if_needed(&mut out, true)?;
                out.extend(bytes);
                i += consumed;
                escape_this = false;
                continue;
            }
            EncapsulationResult::RetrySubsequence { subseq, consumed } => {
                // re-run subseq with embedding disabled
                let mut retry_settings = settings.clone();
                retry_settings.utf8_base64_embed_enabled = false;
                let (retry_bytes, retry_log) = dca_to_utf8(&subseq, &retry_settings)?;
                log.merge(&retry_log);
                unm.flush_if_needed(&mut out, true)?;
                out.extend_from_slice(&retry_bytes);
                i += consumed;
                continue;
            }
        }

        // Normal mapping path
        let mapped = process_mapped_dc(dc, settings, debug, &mut log)?;
        if mapped.is_empty() {
            unm.push(dc);
        } else {
            if settings.dc_basenb_enabled && !unm.is_empty() {
                unm.flush_if_needed(&mut out, false)?;
            }
            out.extend(mapped);
        }

        if escape_this { escape_this = false; }
        i += 1;
    }

    // final flush / close
    unm.flush_if_needed(&mut out, true)?;
    unm.close_if_needed(&mut out);

    Ok((out, log))
}
